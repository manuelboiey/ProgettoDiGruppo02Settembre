"""Strumento RAG per interrogare documenti locali con citazioni.

Questo modulo definisce lo strumento `RAG` basato su FAISS e LangChain per
supportare retrieval-augmented generation sui documenti locali, inclusa la
gestione della persistenza dell'indice e la costruzione della catena di
prompting con citazioni delle fonti.
"""

from __future__ import annotations

import os
from dataclasses import dataclass
from pathlib import Path
from typing import List, ClassVar

from crewai.tools import BaseTool
from dotenv import load_dotenv

from langchain.schema import Document
from langchain_community.vectorstores import FAISS
from langchain.text_splitter import RecursiveCharacterTextSplitter
from langchain_openai import AzureOpenAIEmbeddings

    context_precision,   # "precision@k" sui chunk recuperati

    faithfulness,        # ancoraggio della risposta al contesto
    answer_relevancy,    # pertinenza della risposta vs domanda
    answer_correctness,  # usa questa solo se hai ground_truth
)

@dataclass
class Settings:
    """Parametri di configurazione per vettorizzazione e retrieval."""
    # Persistenza FAISS
    persist_dir: str = "faiss"
    # Text splitting
    chunk_size: int = 1500
    chunk_overlap: int = 300
    # Retriever (MMR)
    search_type: str = "mmr"        # "mmr" o "similarity"
    k: int = 5                      # risultati finali
    fetch_k: int = 20               # candidati iniziali (per MMR)
    mmr_lambda: float = 0.6         # 0 = diversificazione massima, 1 = pertinenza massima
    # Embedding
    hf_model_name: str = "sentence-transformers/all-MiniLM-L6-v2"
    # LM Studio (OpenAI-compatible)
    lmstudio_model_env: str = "LMSTUDIO_MODEL"  # nome del modello in LM Studio, via env var

    def get_embeddings(self, settings: Settings) -> AzureOpenAIEmbeddings:  # noqa: ARG002
        """Inizializza il modello di embedding Azure OpenAI.

        Args:
            settings: Configurazione corrente.

        Returns:
            AzureOpenAIEmbeddings: Oggetto embedding pronto all'uso.

        Raises:
            ValueError: Se le variabili d'ambiente richieste non sono impostate.
        """
        azure_endpoint = os.getenv("AZURE_API_BASE")
        azure_api_key = os.getenv("AZURE_API_KEY")
        azure_api_version = os.getenv("AZURE_API_VERSION")
        
        if not azure_endpoint:
            raise ValueError("AZURE_API_BASE environment variable is not set")
        if not azure_api_key:
            raise ValueError("AZURE_API_KEY environment variable is not set")
        if not azure_api_version:
            raise ValueError("AZURE_API_VERSION environment variable is not set")
        
        return AzureOpenAIEmbeddings(
            model="text-embedding-ada-002",
            azure_endpoint=azure_endpoint,
            api_key=azure_api_key,
            openai_api_version=azure_api_version
        )

    def get_llm_from_lmstudio(self, settings: Settings) -> AzureChatOpenAI:  # noqa: ARG002
        """Inizializza un modello chat su Azure OpenAI.

        Args:
            settings: Configurazione corrente.

        Returns:

            temperature=0.1,
            max_tokens=None,
            timeout=None,
            max_retries=2,
        )

    def load_docs(self, path: str) -> List[Document]:
        """Carica un file di testo come elenco di documenti LangChain.

        Args:
            path: Percorso del file di testo.

        Returns:
            list[Document]: Un singolo documento con `page_content` e `metadata`.
        """
        doc = Path(path).read_text(encoding="utf-8")
        return [Document(page_content=doc, metadata={"source": path})]

    def split_documents(self, docs: List[Document], settings: Settings) -> List[Document]:
        """Esegue lo splitting dei documenti per ottimizzare il retrieval.

        Args:
            docs: Elenco di documenti di input.
            settings: Parametri di splitting (chunk size/overlap).

        Returns:
            list[Document]: Documenti suddivisi in chunk.
        """
        splitter = RecursiveCharacterTextSplitter(
            chunk_size=settings.chunk_size,
            chunk_overlap=settings.chunk_overlap,
            separators=["\n\n", "\n", ". ", "? ", "! ", "; ", ": ", ", ", " ", ""],
        )
        return splitter.split_documents(docs)

    def build_faiss_vectorstore(self, chunks: List[Document], embeddings: AzureOpenAIEmbeddings, persist_dir: str) -> FAISS:
        """Costruisce un indice FAISS e lo salva nella directory di persistenza.
        
        Args:
            chunks: Documenti chunked da indicizzare.
            embeddings: Modello di embedding da usare.
            persist_dir: Directory di output per l'indice.
        
        Returns:
            FAISS: Vector store persistito.
        """
        vs = FAISS.from_documents(
            documents=chunks,
            embedding=embeddings
        )
    
        Path(persist_dir).mkdir(parents=True, exist_ok=True)
        vs.save_local(persist_dir)
        return vs

    def load_or_build_vectorstore(self, settings: Settings, embeddings: AzureOpenAIEmbeddings, docs: List[Document]) -> FAISS:
        """Carica un FAISS esistente oppure lo costruisce da zero.

        Args:
            settings: Configurazione di persistenza e retrieval.
            embeddings: Modello di embedding.
            docs: Documenti sorgente.

        Returns:
            FAISS: Vector store pronto all'uso.
        """
        persist_path = Path(settings.persist_dir)
        index_file = persist_path / "index.faiss"
        meta_file = persist_path / "index.pkl"
    
        if index_file.exists() and meta_file.exists():
            # Dal 2024/2025 molte build richiedono il flag 'allow_dangerous_deserialization' per caricare pkl locali
            return FAISS.load_local(
                settings.persist_dir,
                embeddings,
                allow_dangerous_deserialization=True
            )
    
        chunks = self.split_documents(docs, settings)
        return self.build_faiss_vectorstore(chunks, embeddings, settings.persist_dir)

    def make_retriever(self, vector_store: FAISS, settings: Settings):
        """Configura il retriever FAISS.

        Con `mmr` sono privilegiati risultati meno ridondanti e più coprenti.

        Args:
            vector_store: Indice FAISS.
            settings: Parametri di retrieval.

        Returns:
            Retriever configurato.
        """
        if settings.search_type == "mmr":
            return vector_store.as_retriever(
                search_type="mmr",
                search_kwargs={
                    "k": settings.k,
                    "fetch_k": settings.fetch_k,
                    "lambda_mult": settings.mmr_lambda,
                },
            )
        return vector_store.as_retriever(
            search_type="similarity",
            search_kwargs={"k": settings.k},
        )

    def format_docs_for_prompt(self, docs: List[Document]) -> str:
        """Formatta i documenti in un contesto testuale con citazioni.

        Args:
            docs: Documenti recuperati.

        Returns:
            str: Testo concatenato con citazioni in formato `[source:FILE]`.
        """
        lines: List[str] = []
        for i, d in enumerate(docs, start=1):
            src = d.metadata.get("source", f"doc{i}")
            lines.append(f"[source:{src}] {d.page_content}")
        return "\n\n".join(lines)

    def build_rag_chain(self, llm, retriever):
        """Crea la catena RAG retrieval -> prompt -> LLM con citazioni.

        Args:
            llm: Modello conversazionale da invocare.
            retriever: Retriever per ottenere i documenti.

        Returns:
            Catena eseguibile LCEL che produce una stringa.
        """
        system_prompt = (
            "Sei un assistente esperto. Rispondi in italiano. "
            "Usa esclusivamente il CONTENUTO fornito nel contesto. "
            "Se l'informazione non è presente, dichiara che non è disponibile. "
            "Includi citazioni tra parentesi quadre nel formato [source:...]. "
            "Sii conciso, accurato e tecnicamente corretto."
        )

        prompt = ChatPromptTemplate.from_messages([
            ("system", system_prompt),
            (
                "human",
                "Domanda:\n{question}\n\n"
                "Contesto (estratti selezionati):\n{context}\n\n"
                "Istruzioni:\n"
                "1) Rispondi solo con informazioni contenute nel contesto.\n"
                "2) Cita sempre le fonti pertinenti nel formato [source:FILE].\n"
                "3) Se la risposta non è nel contesto, scrivi: "
                "'Non è presente nel contesto fornito.'",
            ),
        ])
    
        # LCEL: dict -> prompt -> llm -> parser
        chain = (
            {
                "context": retriever | self.format_docs_for_prompt,
                "question": RunnablePassthrough(),
            }
            | prompt
            | llm
            | StrOutputParser()
        )
        return chain

    def rag_answer(self, question: str, chain) -> str:
        """Esegue la catena RAG per una singola domanda.

        Args:
            question: Testo della domanda.
            chain: Catena RAG creata da `build_rag_chain`.

        Returns:
            str: Risposta generata.
        """

    def get_contexts_for_question(self, retriever, question: str, k: int) -> List[str]:
        """Ritorna i testi dei top-k documenti (chunk) usati come contesto."""
        docs = retriever.invoke(question)[:k]
        return [d.page_content for d in docs]

    def build_ragas_dataset(
        self,
        questions: List[str],
        retriever,
        chain,
        *,
        k: int,
        ground_truth: dict[str, str] | None = None,
    ):
        """
        Esegue la pipeline RAG per ogni domanda e costruisce il dataset per Ragas.
        Ogni riga contiene: question, contexts, answer, (opzionale) ground_truth.
        """
        dataset = []
        for q in questions:
            contexts = self.get_contexts_for_question(retriever, q, k)
            answer = chain.invoke(q)

            row = {
                # chiavi richieste da molte metriche Ragas
                "user_input": q,
                "retrieved_contexts": contexts,
                "response": answer,
            }
            if ground_truth and q in ground_truth:
                row["reference"] = ground_truth[q]

            dataset.append(row)
        return dataset

    def _run(self, query: str, run_manager=None) -> str:
        """Punto di ingresso Tool: esegue RAG sulla query fornita.

        Args:
            query: Domanda da porre al sistema RAG.

        Returns:
            str: Risposta sintetica con citazioni.
        """
        load_dotenv()

        embeddings = self.get_embeddings(self.SETTINGS)
        llm = self.get_llm_from_lmstudio(self.SETTINGS)

        vector_store = self.load_or_build_vectorstore(
            self.SETTINGS,
            embeddings,
            self.load_docs('documents\\medicina_generale.txt'),
        )

        retriever = self.make_retriever(vector_store, self.SETTINGS)
        chain = self.build_rag_chain(llm, retriever)
        answer = self.rag_answer(query, chain)

        # 6) Costruisci dataset per Ragas (stessi top-k del tuo retriever)
        dataset = self.build_ragas_dataset(
            questions=[query],
            retriever=retriever,
            chain=chain,
            k=self.SETTINGS.k,
        )

        # 7) Scegli le metriche in base alla disponibilità di ground truth
        metrics = (
            [context_precision, context_recall, faithfulness, answer_relevancy, answer_correctness]
            if all("reference" in row for row in dataset)
            else [faithfulness, answer_relevancy]
        )

        # 8) Esegui la valutazione con il TUO LLM e le TUE embeddings
        evaluate(
            dataset=EvaluationDataset.from_list(dataset),
            metrics=metrics,
            llm=llm,
            embeddings=embeddings,
        ).to_pandas().to_csv("ragas_results.csv", index=False)
        print("Salvato: ragas_results.csv")

        return answer
